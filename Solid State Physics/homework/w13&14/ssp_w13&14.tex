\begin{problem}
试证在磁场中运动的布洛赫电子，在 $k$ 空间中轨迹面积 $S_n$ 和在 $r$ 空间中的轨迹面积 $A_n$ 之间的关系为
\[
A_n=\left(\frac{\hbar c}{qB}\right)^2S_n.
\]
\end{problem}

\begin{solution}
半经典运动方程为
\[
\hbar \dot{\mathbf k}
=
-\frac{q}{c}\mathbf v\times\mathbf B .
\]

又有
\[
\mathbf v
=
\frac1{\hbar}\nabla_k E.
\]

因此
\[
d\mathbf r
=
\mathbf v\,dt
=
-\frac{\hbar c}{qB}
\hat z\times d\mathbf k.
\]

故 $r$ 空间轨道与 $k$ 空间轨道相似，相应长度满足

\[
dl_r
=
\frac{\hbar c}{qB}dl_k.
\]

面积按长度平方缩放：

\[
A_n
=
\left(
\frac{\hbar c}{qB}
\right)^2
S_n.
\]

即得证。
\end{solution}

\begin{problem}
根据自由电子模型计算钾的德哈斯—范阿尔芬效应中的周期
\[
\Delta\!\left(\frac1B\right).
\]

对于 $B=1\,{\rm T}$，求电子在实空间轨道面积。
\end{problem}

\begin{solution}
Onsager量子化条件：

\[
S(E_F)
=
\frac{2\pi qB}{\hbar c}
\left(n+\gamma\right).
\]

因此振荡周期为

\[
\Delta\!\left(\frac1B\right)
=
\frac{2\pi q}{\hbar c\,S_F}.
\]

自由电子球面：

\[
S_F=\pi k_F^2.
\]

其中

\[
k_F=(3\pi^2n)^{1/3}.
\]

代入钾的电子浓度

\[
n\simeq1.4\times10^{28}\ {\rm m^{-3}}
\]

得到

\[
k_F\simeq7.5\times10^9\ {\rm m^{-1}},
\]

\[
S_F\simeq1.8\times10^{20}\ {\rm m^{-2}}.
\]

故

\[
\boxed{
\Delta\!\left(\frac1B\right)
\simeq3.7\times10^{-5}\ {\rm T^{-1}}
}
\]

当 $B=1\,{\rm T}$ 时

\[
A
=
\left(
\frac{\hbar}{eB}
\right)^2S_F
\]

得

\[
\boxed{
A\simeq8\times10^{-12}\ {\rm m^2}.
}
\]
\end{solution}

\begin{problem}
$^3$He 的自旋为 $1/2$，是费米子。液体 $^3$He 在绝对零度附近的密度约为
\[
0.081\,{\rm g\,cm^{-3}}.
\]

计算费米能 $E_F$ 和费米温度 $T_F$。
\end{problem}

\begin{solution}
粒子浓度：

\[
n
=
\frac{\rho N_A}{M}
=
\frac{0.081\times10^3}{3}
N_A
\simeq1.63\times10^{28}\ {\rm m^{-3}}.
\]

费米波矢：

\[
k_F=(3\pi^2n)^{1/3}
\simeq7.8\times10^9\ {\rm m^{-1}}.
\]

费米能：

\[
E_F
=
\frac{\hbar^2k_F^2}{2m_{^3{\rm He}}}
\simeq8.5\times10^{-23}\ {\rm J}.
\]

即

\[
\boxed{
E_F\simeq5.3\times10^{-4}\ {\rm eV}
}
\]

费米温度：

\[
T_F=\frac{E_F}{k_B}
\]

故

\[
\boxed{
T_F\simeq6.2\ {\rm K}
}
\]
\end{solution}

\begin{problem}
在低温下金属钾的摩尔电子热容实验结果为

\[
C_e=2.08\,T
\ {\rm mJ\,mol^{-1}K^{-1}}.
\]

在自由电子气模型下估算钾的费米温度 $T_F$ 和费米面上的态密度 $N(E_F)$。
\end{problem}

\begin{solution}
自由电子模型：

\[
C_e=\gamma T,
\]

且

\[
\gamma
=
\frac{\pi^2}{2}
\frac{R}{T_F}.
\]

由实验值

\[
\gamma
=
2.08\times10^{-3}
{\rm J\,mol^{-1}K^{-2}}
\]

得到

\[
\boxed{
T_F
\simeq2.0\times10^4\ {\rm K}
}
\]

费米面态密度满足

\[
\gamma
=
\frac{\pi^2}{3}
k_B^2N(E_F).
\]

故

\[
\boxed{
N(E_F)
=
\frac{3\gamma}{\pi^2k_B^2}
\simeq3.3\times10^{42}
\ {\rm J^{-1}mol^{-1}}
}
\]
\end{solution}

\begin{problem}
若把银看成具有球形费米面的单价金属，计算：

\begin{enumerate}
\item 费米能和费米温度；
\item 费米球半径；
\item 费米速度；
\item 费米球面的横截面积；
\item 在室温及低温时电子的平均自由程。
\end{enumerate}

已知：

\[
\rho_m=10.5\ {\rm g\,cm^{-3}},
\qquad
M=107.87,
\]

\[
\rho(295{\rm K})
=
1.61\times10^{-6}
\Omega\cdot{\rm cm},
\]

\[
\rho(20{\rm K})
=
0.038\times10^{-6}
\Omega\cdot{\rm cm}.
\]
\end{problem}

\begin{solution}

银为单价金属，

\[
n=\frac{\rho_mN_A}{M}
\simeq5.86\times10^{28}\ {\rm m^{-3}}.
\]

费米波矢：

\[
k_F=(3\pi^2n)^{1/3}
\]

\[
\boxed{
k_F
=
1.20\times10^{10}\ {\rm m^{-1}}
}
\]

费米能：

\[
E_F
=
\frac{\hbar^2k_F^2}{2m}
\]

\[
\boxed{
E_F
=
5.48\ {\rm eV}
}
\]

费米温度：

\[
T_F=\frac{E_F}{k_B}
\]

\[
\boxed{
T_F
=
6.35\times10^4\ {\rm K}
}
\]

费米速度：

\[
v_F=\frac{\hbar k_F}{m}
\]

\[
\boxed{
v_F
=
1.39\times10^6\ {\rm m/s}
}
\]

费米球最大横截面积：

\[
S_F=\pi k_F^2
\]

\[
\boxed{
S_F
=
4.52\times10^{20}\ {\rm m^{-2}}
}
\]

由 Drude 模型

\[
\rho=\frac{m}{ne^2\tau},
\qquad
l=v_F\tau.
\]

得到：

\[
\boxed{
l_{295K}
=
5.24\times10^{-8}\ {\rm m}
=
52.4\ {\rm nm}
}
\]

\[
\boxed{
l_{20K}
=
2.22\times10^{-6}\ {\rm m}
=
2.22\ \mu{\rm m}
}
\]

\end{solution}

\begin{problem}
设 $N$ 个电子组成简并自由电子气，体积为 $V$，证明 $T=0$ K 时：

\begin{enumerate}
\item 每个电子平均能量
\[
\bar U=\frac35E_F;
\]
\item 自由电子气压强满足
\[
pV=\frac23U.
\]
\end{enumerate}
\end{problem}

\begin{solution}
总能量：

\[
U
=
\int_0^{E_F}
E\,g(E)\,dE.
\]

自由电子态密度：

\[
g(E)
=
C\sqrt E.
\]

因此

\[
N
=
C\int_0^{E_F}
E^{1/2}dE
=
\frac23CE_F^{3/2},
\]

\[
U
=
C\int_0^{E_F}
E^{3/2}dE
=
\frac25CE_F^{5/2}.
\]

于是

\[
\frac{U}{N}
=
\frac35E_F.
\]

即

\[
\boxed{
\bar U=\frac35E_F.
}
\]

又因为

\[
U=\frac35NE_F
\propto V^{-2/3}.
\]

利用热力学关系

\[
p=-\left(\frac{\partial U}{\partial V}\right)_N
\]

可得

\[
p=\frac23\frac{U}{V}.
\]

故

\[
\boxed{
pV=\frac23U.
}
\]
\end{solution}
